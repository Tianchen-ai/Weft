cmake_minimum_required(VERSION 3.20)
project(WeftNativeConsumers LANGUAGES C CXX)
find_package(Weft CONFIG REQUIRED)
add_executable(native-sdk native_sdk.cpp)
target_compile_features(native-sdk PRIVATE cxx_std_17)
target_link_libraries(native-sdk PRIVATE Weft::Runtime)
add_executable(native-c native_sdk.c)
target_link_libraries(native-c PRIVATE Weft::Runtime m)
if(DEFINED LLAMA_SOURCE AND DEFINED LLAMA_LIBRARIES)
  add_executable(ggml-nodes ggml_nodes.cpp)
  target_compile_features(ggml-nodes PRIVATE cxx_std_17)
  target_include_directories(ggml-nodes PRIVATE "${LLAMA_SOURCE}/ggml/include")
  target_link_directories(ggml-nodes PRIVATE "${LLAMA_LIBRARIES}")
  target_link_libraries(ggml-nodes PRIVATE ggml ggml-cpu ggml-base)
  if(WEFT_GGML_OBSERVE)
    target_compile_definitions(ggml-nodes PRIVATE WEFT_GGML_OBSERVE)
    target_include_directories(ggml-nodes PRIVATE "${Weft_LLAMA_CPP_DIR}/include")
    target_link_libraries(ggml-nodes PRIVATE Weft::Runtime)
  endif()
endif()
